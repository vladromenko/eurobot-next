cmake_minimum_required(VERSION 3.22)
project(eurobot_hardware LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_srvs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(eurobot_interfaces REQUIRED)
# Repository-local common wire/core implementation is the ONLY protocol implementation.
get_filename_component(EUROBOT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_library(eurobot_transport STATIC src/transport.cpp "${EUROBOT_ROOT}/firmware/src/core.cpp")
target_include_directories(eurobot_transport PUBLIC include "${EUROBOT_ROOT}/firmware/include")
target_compile_options(eurobot_transport PRIVATE -Wall -Wextra -Wpedantic)
add_executable(eurobot_bridge src/bridge.cpp)
target_link_libraries(eurobot_bridge PRIVATE eurobot_transport rclcpp::rclcpp rclcpp_action::rclcpp_action)
target_compile_options(eurobot_bridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
# Lyrical uses aggregate imported targets; Jazzy exports explicit C++ type support.
# Both carry generated headers and linked typesupport without removed ament macros.
foreach(interface_package IN ITEMS rcl_interfaces std_srvs geometry_msgs sensor_msgs diagnostic_msgs eurobot_interfaces)
  if(TARGET ${interface_package}::${interface_package})
    target_link_libraries(eurobot_bridge PRIVATE ${interface_package}::${interface_package})
  elseif(TARGET ${interface_package}::${interface_package}__rosidl_typesupport_cpp)
    target_link_libraries(eurobot_bridge PRIVATE ${interface_package}::${interface_package}__rosidl_typesupport_cpp)
  else()
    message(FATAL_ERROR "No C++ interface target exported by ${interface_package}")
  endif()
endforeach()
install(TARGETS eurobot_bridge DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  find_package(Threads REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  add_executable(transport_test test/transport_test.cpp)
  target_link_libraries(transport_test eurobot_transport Threads::Threads)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_libraries(transport_test util)
  endif()
  add_test(NAME transport_test COMMAND transport_test)
  ament_add_pytest_test(gripper_action_test test/test_gripper_action.py
    TIMEOUT 60
    ENV "BRIDGE_EXECUTABLE=$<TARGET_FILE:eurobot_bridge>" "ROS_DOMAIN_ID=173")
endif()
ament_package()
