cmake_minimum_required(VERSION 3.25)
project(EUROBOT_NEXT VERSION 0.1.0 LANGUAGES C CXX ASM)
option(EUROBOT_EXPERIMENTAL_CXX26 "Probe full build in experimental C++26 mode" OFF)
if(EUROBOT_EXPERIMENTAL_CXX26)
 set(CMAKE_CXX_STANDARD 26)
else()
 set(CMAKE_CXX_STANDARD 23)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(EUROBOT_STM32 "Build Cortex-M4F firmware" OFF)
option(EUROBOT_SANITIZERS "Host ASan+UBSan" OFF)
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("#include <span>
#include <array>
#if __cplusplus < 202100L
#error C++23 required
#endif
int main(){std::array<int,2> a{}; std::span<int> s{a}; return s[0];}" EUROBOT_CXX23_WORKS)
if(NOT EUROBOT_CXX23_WORKS)
 message(FATAL_ERROR "Compiler must support C++23 and std::span")
endif()
add_library(eurobot_core STATIC firmware/src/core.cpp firmware/src/devices.cpp firmware/src/gripper.cpp firmware/src/application.cpp)
target_include_directories(eurobot_core PUBLIC firmware/include)
target_compile_options(eurobot_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(EUROBOT_STM32)
 add_subdirectory(firmware)
else()
 enable_testing()
 if(EUROBOT_SANITIZERS)
  target_compile_options(eurobot_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(eurobot_core PUBLIC -fsanitize=address,undefined)
 endif()
 foreach(name core device gripper application)
  add_executable(eurobot_${name}_tests tests/${name}_tests.cpp)
  target_link_libraries(eurobot_${name}_tests PRIVATE eurobot_core)
  target_compile_options(eurobot_${name}_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME ${name} COMMAND eurobot_${name}_tests)
 endforeach()
 add_executable(eurobot_runtime_tests tests/runtime_tests.cpp)
 target_include_directories(eurobot_runtime_tests PRIVATE tests/fakes)
 target_link_libraries(eurobot_runtime_tests PRIVATE eurobot_core)
 target_compile_options(eurobot_runtime_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
 add_test(NAME firmware_runtime COMMAND eurobot_runtime_tests)
 find_package(GTest REQUIRED)
 add_executable(eurobot_profile_tests tests/profile_tests.cpp tests/sensor_wire_tests.cpp tests/input_tests.cpp tests/gripper_wire_tests.cpp tests/maintenance_wire_tests.cpp)
 target_link_libraries(eurobot_profile_tests PRIVATE eurobot_core GTest::gtest_main)
 include(GoogleTest)
 gtest_discover_tests(eurobot_profile_tests)
 find_package(Threads REQUIRED)
 add_executable(eurobot_transport_tests ros2_ws/src/eurobot_hardware/src/transport.cpp ros2_ws/src/eurobot_hardware/test/transport_test.cpp)
 target_include_directories(eurobot_transport_tests PRIVATE ros2_ws/src/eurobot_hardware/include)
 target_link_libraries(eurobot_transport_tests PRIVATE eurobot_core Threads::Threads)
 if(UNIX AND NOT APPLE)
  target_link_libraries(eurobot_transport_tests PRIVATE util)
 endif()
 add_test(NAME serial_transport COMMAND eurobot_transport_tests)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tests/flash_test.py)
  add_test(NAME flash_safety COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/flash_test.py)
 endif()
 add_test(NAME source_policy COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_policy.py)
 add_test(NAME mac_board_tool COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/board_test_test.py)
endif()
